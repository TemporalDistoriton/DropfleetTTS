--[[
    Ship Picker Script - Industrial Faction
    ---------------------------------
    A ship card selection tool that saves card data and displays them in a searchable
    XML UI interface. Players can browse ships by tonnage category, search by name,
    select multiple cards, and spawn them to their play area.

    Features:
    - Setup Mode: Select cards to save using box-selection
    - Card JSON saving: Captures card data including nested scripts
    - View Ships: Opens XML UI in player's Global UI showing all saved cards
    - Tonnage Tabs: Light, Medium, Heavy, Colossal (based on baseScale in card script)
    - Search Function: Filter cards by name
    - Multi-select: Click cards to highlight green for selection
    - Spawn to Player Area: Spawns selected cards in a line at player's position
--]]

-- Saves data by embedding it directly into the Lua script
function updateSave()
    local script = self.getLuaScript()

    -- Find where the embedded data section starts
    local dataStartPattern = "\n%-%-EMBEDDED_DATA_START"
    local dataStart = script:find(dataStartPattern)

    if dataStart then
        script = script:sub(1, dataStart - 1)
    end

    -- Build the embedded data section
    local embeddedData = "\n--EMBEDDED_DATA_START\n"
    embeddedData = embeddedData .. "savedShipCards = {\n"

    for i, card in ipairs(savedShipCards) do
        embeddedData = embeddedData .. "  {\n"
        embeddedData = embeddedData .. "    name = [====[" .. (card.name or "Unknown") .. "]====],\n"
        embeddedData = embeddedData .. "    imageURL = [====[" .. (card.imageURL or "") .. "]====],\n"
        embeddedData = embeddedData .. "    tonnage = [====[" .. (card.tonnage or "Light") .. "]====],\n"
        embeddedData = embeddedData .. "    baseScale = " .. (card.baseScale or 1) .. ",\n"
        if card.json then
            embeddedData = embeddedData .. "    json = [====[" .. card.json .. "]====],\n"
        end
        embeddedData = embeddedData .. "  },\n"
    end

    embeddedData = embeddedData .. "}\n\n"
    embeddedData = embeddedData .. "fresh = " .. tostring(fresh) .. "\n"

    script = script .. embeddedData
    self.setLuaScript(script)
end

-- Called when the object is loaded
function onLoad(saved_data)
    selfGUID = self.getGUID()

    -- Initialize variables if they don't exist
    if savedShipCards == nil then
        savedShipCards = {}
    end
    if fresh == nil then
        fresh = true
    end

    -- Update fresh status based on data
    if #savedShipCards > 0 then
        fresh = false
    end

    -- Initialize player state tables
    playerSelections = {}
    playerCardIndices = {}
    playerSearchFilters = {}
    playerActiveTabs = {}
    playerSpawnCounts = {}

    -- Clean loaded data
    for _, card in ipairs(savedShipCards) do
        if card.json then card.json = stripBrackets(card.json) end
        if card.imageURL then card.imageURL = stripBrackets(card.imageURL) end
    end

    -- Show appropriate buttons
    if fresh then
        createSetupButton()
    else
        createMainButtonsWithList()
    end
end

-- Enters setup mode
function buttonClick_setup()
    savedShipCardsBackup = duplicateTable(savedShipCards)
    createSetupActionButtons()
    broadcastToAll("Setup Mode: Box-select ship cards, then click Add/Remove", {1, 1, 1})
end

-- Cancel setup
function buttonClick_cancel()
    savedShipCards = savedShipCardsBackup

    if #savedShipCards == 0 then
        createSetupButton()
    else
        createMainButtonsWithList()
    end

    broadcastToAll("Setup Canceled", {1, 1, 1})
end

-- Add selected cards to memory
function buttonClick_add(obj, player_color)
    local selectedObjs = Player[player_color].getSelectedObjects()
    local count = 0

    for _, selectedObj in ipairs(selectedObjs) do
        if selectedObj and not selectedObj.isDestroyed() and selectedObj ~= self then
            -- Check if it's a card or tile
            if selectedObj.tag == "Card" or selectedObj.tag == "Tile" then
                -- Check if already saved (by name comparison)
                local cardName = selectedObj.getName()
                local alreadyExists = false

                for _, saved in ipairs(savedShipCards) do
                    if saved.name == cardName then
                        alreadyExists = true
                        break
                    end
                end

                if not alreadyExists then
                    -- Extract card data
                    local cardData = extractCardData(selectedObj)
                    if cardData then
                        table.insert(savedShipCards, cardData)
                        count = count + 1
                    end
                end
            end
        end
    end

    if count > 0 then
        fresh = false
        -- Sort cards alphabetically
        sortShipCards()
        updateSave()
        broadcastToAll(count .. " ship card(s) added to " .. CONFIG.FACTION_NAME .. " Ship Picker", {0.2, 1, 0.2})
    else
        broadcastToAll("No new cards to add (select Card objects)", {1, 0.5, 0.5})
    end

    createMainButtonsWithList()
end

-- Remove selected cards from memory
function buttonClick_remove(obj, player_color)
    local selectedObjs = Player[player_color].getSelectedObjects()
    local count = 0

    local newSavedCards = {}

    for _, saved in ipairs(savedShipCards) do
        local shouldRemove = false

        for _, selectedObj in ipairs(selectedObjs) do
            if selectedObj and not selectedObj.isDestroyed() then
                if selectedObj.getName() == saved.name then
                    shouldRemove = true
                    count = count + 1
                    break
                end
            end
        end

        if not shouldRemove then
            table.insert(newSavedCards, saved)
        end
    end

    savedShipCards = newSavedCards

    if #savedShipCards == 0 then
        fresh = true
        createSetupButton()
    else
        createMainButtonsWithList()
    end

    updateSave()
    broadcastToAll(count .. " ship card(s) removed from " .. CONFIG.FACTION_NAME .. " Ship Picker", {1, 0.5, 0.5})
end

-- Reset all saved data
function buttonClick_reset()
    fresh = true
    savedShipCards = {}
    savedShipCardsBackup = {}
    playerSelections = {}
    playerSearchFilters = {}
    playerActiveTabs = {}
    playerSpawnCounts = {}

    createSetupButton()
    updateSave()
    broadcastToAll(CONFIG.FACTION_NAME .. " Ship Picker Reset", {1, 1, 1})
end

--==============================================================================
-- CARD DATA EXTRACTION
--==============================================================================

-- Extracts card data from a card object
function extractCardData(cardObj)
    if not cardObj then return nil end

    -- Only allow Tile objects
    if cardObj.tag ~= "Tile" then
        broadcastToAll("Ignored object: Only Tile objects can be saved.", {1, 0.5, 0})
        return nil
    end

    local cardData = {
        name = cardObj.getName() or "Unknown Ship",
        json = nil,
        imageURL = "",
        tonnage = "Light",
        baseScale = 1,
    }

    -- Capture the exact JSON of the object (Tile)
    -- We use this exactly as saved for spawning
    cardData.json = cardObj.getJSON()

    -- Extract Image URL for the UI listing (from CustomObject)
    local customObj = cardObj.getCustomObject()
    if customObj and customObj.image then
        cardData.imageURL = customObj.image
    end

    -- Tonnage detection from Description (check for specific ship types)
    local desc = cardObj.getDescription() or ""
    local descLower = string.lower(desc)
    local name = cardObj.getName() or ""
    nameLower = string.lower(name)

    -- Check both description and name for ship type
    local text = descLower .. " " .. nameLower

    -- Ordered from largest to smallest for proper detection
    if string.find(text, "dreadnought") or string.find(text, "dreadnaught") then
        cardData.tonnage = "Dreadnaught"
    elseif string.find(text, "super battleship") or string.find(text, "superbattleship") then
        cardData.tonnage = "Super Battleship"
    elseif string.find(text, "supercarrier") then
        cardData.tonnage = "Supercarrier"
    elseif string.find(text, "battleship") then
        cardData.tonnage = "Battleship"
    elseif string.find(text, "battlecruiser") then
        cardData.tonnage = "Battlecruiser"
    elseif string.find(text, "troopship") then
        cardData.tonnage = "Troopship"
    elseif string.find(text, "heavy cruiser") then
        cardData.tonnage = "Heavy Cruiser"
    elseif string.find(text, "cruiser") then
        cardData.tonnage = "Cruiser"
    elseif string.find(text, "light cruiser") then
        cardData.tonnage = "Light Cruiser"
    elseif string.find(text, "runner") then
        cardData.tonnage = "Runner"
    elseif string.find(text, "destroyer") then
        cardData.tonnage = "Destroyer"
    elseif string.find(text, "cutter") then
        cardData.tonnage = "Cutter"
    elseif string.find(text, "monitor") then
        cardData.tonnage = "Monitor"
    elseif string.find(text, "carrier") then
        cardData.tonnage = "Carrier"
    elseif string.find(text, "frigate") then
        cardData.tonnage = "Frigate"
    elseif string.find(text, "lighter") then
        cardData.tonnage = "Lighter"
    elseif string.find(text, "corvette") then
        cardData.tonnage = "Corvette"
    elseif string.find(text, "cell") then
        cardData.tonnage = "Cell"
    else
        cardData.tonnage = "Other" -- Default for unrecognized types
    end

    -- Attempt to find baseScale in the Lua script for Tonnage Category sorting
    -- (This preserves the existing logic for categorizing ships in the UI)
    local script = cardObj.getLuaScript()
    if script then
        local scale = script:match("baseScale%s*=%s*([%d%.]+)")
        if scale then
            cardData.baseScale = tonumber(scale)
        end
    end

    return cardData
end


-- Sorts ship cards alphabetically by name
function sortShipCards()
    table.sort(savedShipCards, function(a, b)
        return (a.name or "") < (b.name or "")
    end)
end

-- Opens the ship viewer UI for a player
function buttonClick_viewShips(obj, player_color)
    if #savedShipCards == 0 then
        broadcastToColor("No ships saved! Use Setup to add ship cards.", player_color, {1, 0.5, 0.5})
        return
    end

    if not playerSearchFilters[player_color] then playerSearchFilters[player_color] = "" end
    if not playerActiveTabs[player_color] then playerActiveTabs[player_color] = "All" end
    if not playerCardIndices[player_color] then playerCardIndices[player_color] = 1 end

    -- Clear object buttons before showing UI
    self.clearButtons()

    -- Generate and display the UI
    showShipPickerUI(player_color)
end

-- Generates and shows the ship picker XML UI for a player
function showShipPickerUI(playerColor)
    local shipPickerXML = generateShipPickerXML(playerColor)
    self.UI.setXml(shipPickerXML)
end

    local filteredCards = getFilteredCards(activeTab, searchFilter)
    local totalCards = #filteredCards

    if totalCards == 0 then
        currentIndex = 0
    elseif currentIndex > totalCards then
        currentIndex = 1
        playerCardIndices[playerColor] = 1
    elseif currentIndex < 1 then
        currentIndex = 1
        playerCardIndices[playerColor] = 1

-- Spawns selected cards for a player at their play area
function spawnCardsForPlayer(playerColor, cardIndices)
    if not playerSpawnCounts then playerSpawnCounts = {} end

    local spawnConfig = CONFIG.PLAYER_SPAWN_POSITIONS[playerColor]

    if not spawnConfig then
        -- Default spawn position if player color not configured
        spawnConfig = {
            position = {x = 0, y = 1, z = 0},
            rotation = {x = 0, y = 0, z = 0},
            direction = {x = 1, y = 0, z = 0},
        }
        broadcastToColor("Warning: No spawn position configured for " .. playerColor .. ", using center.", playerColor, {1, 1, 0})
    end

    local startPos = spawnConfig.position
    local rotation = spawnConfig.rotation
    local direction = spawnConfig.direction
    local spacing = CONFIG.CARD_SPAWN_SPACING

    -- Get persistent spawn count for this player
    local currentCount = playerSpawnCounts[playerColor] or 0

    for i, cardIndex in ipairs(cardIndices) do
        local card = savedShipCards[cardIndex]
        if card and card.json then
            -- Calculate spawn position based on cumulative count
            local offset = currentCount * spacing
            local spawnPos = {
                x = startPos.x + (direction.x * offset),
                y = startPos.y + (direction.y * offset),
                z = startPos.z + (direction.z * offset),
            }

            -- Spawn the card
            spawnObjectJSON({
                json = card.json,
                position = spawnPos,
                rotation = rotation,
                sound = true,
            })

            -- Increment count for next card
            currentCount = currentCount + 1
        end
    end

    -- Update persistent count
    playerSpawnCounts[playerColor] = currentCount
end

-- Duplicates a table (shallow copy for simple tables, recursive for nested)
function duplicateTable(oldTable)
    if type(oldTable) ~= "table" then return oldTable end

    local newTable = {}
    for k, v in pairs(oldTable) do
        if type(v) == "table" then
            newTable[k] = duplicateTable(v)
        else
            newTable[k] = v
        end
    end
    return newTable
end

-- Rounds a number to the specified decimal places
function round(num, dec)
    local mult = 10 ^ (dec or 0)
    return math.floor(num * mult + 0.5) / mult
end

-- Strips surrounding brackets [[...]] from a string if present
function stripBrackets(str)
    if not str then return "" end
    -- Check for [=[ ... ]=] style first
    local s, e = str:find("^%[=%[.*%]=%]$")
    if s then return str:sub(4, -4) end

    -- Check for [[ ... ]] style
    s, e = str:find("^%[%[.*%]%]$")
    if s then return str:sub(3, -3) end

    return str
end

-- Parses a fleet list and spawns the ships
function parseAndSpawnList(listText, playerColor)
    local lines = splitIntoLines(listText)
    local shipEntries = extractShipEntries(lines)

    if #shipEntries == 0 then
        broadcastToColor("No ships found in the list. Make sure the format is correct.", playerColor, {1, 0.5, 0})
        return
    end

    local spawnedCount = 0
    local notFoundList = {}

    for _, entry in ipairs(shipEntries) do
        local quantity = entry.quantity
        local shipName = entry.name
        local matchedCards = findMatchingCards(shipName)

        if #matchedCards > 0 then
            local cardIndex = matchedCards[1]
            for i = 1, quantity do
                spawnCardsForPlayer(playerColor, {cardIndex})
                spawnedCount = spawnedCount + 1
            end
        else
            table.insert(notFoundList, shipName)
        end
    end

    if spawnedCount > 0 then
        broadcastToColor("Spawned " .. spawnedCount .. " ship cards from list.", playerColor, {0.2, 1, 0.3})
    end

    if #notFoundList > 0 then
        local notFoundMsg = "Warning: Could not find cards for: "
        for i, name in ipairs(notFoundList) do
            if i > 1 then notFoundMsg = notFoundMsg .. ", " end
            notFoundMsg = notFoundMsg .. name
            if i >= 5 then
                notFoundMsg = notFoundMsg .. " (and more...)"
                break
            end
        end
        broadcastToColor(notFoundMsg, playerColor, {1, 1, 0})
    end
end

-- Splits text into lines using character-by-character processing
function splitIntoLines(text)
    local lines = {}
    local currentLine = ""
    for i = 1, #text do
        local char = text:sub(i, i)
        if char == "\n" or char == "\r" then
            if #currentLine > 0 then
                table.insert(lines, currentLine)
                currentLine = ""
            end
        else
            currentLine = currentLine .. char
        end
    end
    if #currentLine > 0 then
        table.insert(lines, currentLine)
    end
    return lines
end

-- Removes line number prefix from a line (e.g., "11. ")
function removeLineNumber(line)
    local dotPos = findChar(line, ".")
    if dotPos and dotPos < 5 then
        local beforeDot = trimString(line:sub(1, dotPos - 1))
        if isNumeric(beforeDot) then
            return trimString(line:sub(dotPos + 1))
        end
    end
    return line
end

-- Extracts ship entries from lines
function extractShipEntries(lines)
    local entries = {}
    for _, line in ipairs(lines) do
        line = trimString(line)
        if #line > 0 and not isHeaderLine(line) then
            local entry = parseShipLine(line)
            if entry then
                table.insert(entries, entry)
            end
        end
    end
    return entries
end

-- Checks if a line is a header/section line
function isHeaderLine(line)
    if #line == 0 then return true end
    if line:sub(1, 1) == "#" then return true end

    local lowerLine = line:lower()

    -- Check for configuration keywords
    if contains(lowerLine, "configuration") or
       contains(lowerLine, "game size") or
       contains(lowerLine, "fleet ++") or
       contains(lowerLine, "famous admirals") then
        return true
    end

    -- Check for section headers like "Admirals [25 pts]" or "Heavy Groups [180 pts]"
    if (contains(lowerLine, "admirals") or contains(lowerLine, "groups")) and
       contains(lowerLine, "[") and contains(lowerLine, "pts]") then
        -- But make sure it's not a ship line (ships have bullets or quantity)
        if not contains(lowerLine, "•") and not contains(line, "x ") then
            return true
        end
    end

    return false
end

-- Parses a single ship line and extracts name and quantity
function parseShipLine(line)
    line = trimString(line)
    if #line == 0 then return nil end

    -- Remove line number if present
    line = removeLineNumber(line)
    line = trimString(line)

    -- Check if this is a "group header" line (ends with colon, no bullet, no quantity)
    local colonPos = findChar(line, ":")
    local bulletPresent = (line:sub(1, 1) == "•" or line:sub(1, 1) == "-" or line:sub(1, 1) == "*")

    -- If line ends with ":" and has no bullet and no "x " pattern, it's likely a group header
    if colonPos == #line and not bulletPresent then
        -- Check if there's a quantity pattern
        local hasQuantity = false
        for i = 1, 6 do
            if i < #line and line:sub(i, i) == "x" and line:sub(i+1, i+1) == " " then
                hasQuantity = true
                break
            end
        end
        if not hasQuantity then
            return nil  -- Skip group headers
        end
    end

    -- Remove bullet point if present
    if bulletPresent then
        line = trimString(line:sub(2))
    end

    -- Look for quantity prefix like "2x"
    local quantity = 1
    local xPos = findChar(line, "x")
    if xPos and xPos < 6 then
        local beforeX = trimString(line:sub(1, xPos - 1))
        if isNumeric(beforeX) then
            quantity = tonumber(beforeX) or 1
            line = trimString(line:sub(xPos + 2))
        end
    end

    -- Extract ship name (before bracket or colon)
    local bracketPos = findChar(line, "[")
    colonPos = findChar(line, ":")

    local nameEnd = #line
    if bracketPos and (not colonPos or bracketPos < colonPos) then
        nameEnd = bracketPos - 1
    elseif colonPos then
        nameEnd = colonPos - 1
    end

    local shipName = trimString(line:sub(1, nameEnd))

    -- Skip invalid entries
    if #shipName < 3 then return nil end

    local lowerName = shipName:lower()

    -- Skip admiral/captain lines
    if contains(lowerName, "lvl ") or contains(lowerName, "captain") or contains(lowerName, "admiral") then
        return nil
    end

    -- Skip lines that look like the list title (contain " - " and end with " -")
    if contains(shipName, " - ") and (shipName:sub(-2) == " -" or contains(shipName, "pts]")) then
        return nil
    end

    return {name = shipName, quantity = quantity}
end

-- Finds matching cards by ship name
function findMatchingCards(searchName)
    local matches = {}
    local searchLower = searchName:lower()

    for i, card in ipairs(savedShipCards) do
        local cardName = (card.name or "")
        local cardLower = cardName:lower()

        -- Exact match
        if cardLower == searchLower then
            table.insert(matches, i)
        -- Card name contained in search name
        elseif contains(searchLower, cardLower) then
            table.insert(matches, i)
        -- Search name contained in card name
        elseif contains(cardLower, searchLower) then
            table.insert(matches, i)
        end
    end

    return matches
end

-- Utility: Trims whitespace from both ends of a string
function trimString(str)
    if not str then return "" end
    local startPos = 1
    while startPos <= #str do
        local char = str:sub(startPos, startPos)
        if char ~= " " and char ~= "\t" and char ~= "\r" and char ~= "\n" then
            break
        end
        startPos = startPos + 1
    end
    local endPos = #str
    while endPos >= 1 do
        local char = str:sub(endPos, endPos)
        if char ~= " " and char ~= "\t" and char ~= "\r" and char ~= "\n" then
            break
        end
        endPos = endPos - 1
    end
    if startPos > endPos then return "" end
    return str:sub(startPos, endPos)
end

-- Utility: Finds the first occurrence of a character
function findChar(str, char)
    for i = 1, #str do
        if str:sub(i, i) == char then
            return i
        end
    end
    return nil
end

-- Utility: Checks if a string contains a substring
function contains(str, substr)
    if not str or not substr then return false end
    if #substr > #str then return false end
    for i = 1, #str - #substr + 1 do
        local match = true
        for j = 1, #substr do
            if str:sub(i + j - 1, i + j - 1) ~= substr:sub(j, j) then
                match = false
                break
            end
        end
        if match then return true end
    end
    return false
end

-- Utility: Checks if a string is numeric
function isNumeric(str)
    if not str or #str == 0 then return false end
    for i = 1, #str do
        local char = str:sub(i, i)
        if char < "0" or char > "9" then
            return false
        end
    end
    return true
end
